\pbox{0.7}{2.0}{8.35}{8.2}{ublue}{%
 \hd{ublue}{3. Uneven Cash Flows}
 Money received is different every year. Bring each amount back to today.
 \fmt{$PV=\dfrac{CF_1}{(1+i)}+\dfrac{CF_2}{(1+i)^2}$}

\pbox{18.3}{13.33}{8.35}{5.37}{ured}{%
 \hd{ured}{1. Decision under Risk}

 \eg{Eg: 50\%: Rs 100, 50\%: Rs 40 $\Rightarrow$ EMV = Rs 70}}

\end{scope}

% VI2 Decision under uncertainty   (x=26.95)
\pbox{26.95}{13.33}{8.35}{5.37}{ured}{%
 \hd{ured}{2. Decision under Uncertainty}
 Chances are unknown. Use a rule:\\
 $\bullet$ \textbf{Maximax}: best of the best\\
 $\bullet$ \textbf{Maximin}: best of the worst\\[4pt]
 \renewcommand{\arraystretch}{1.1}%
 \hspace{0.3in}\begin{tabular}{|l|c|c|}\hline
 Payoff & Boom & Slump\\\hline
 Stock & 80 & $-20$\\\hline
 Bond & 40 & 20\\\hline
 \end{tabular}\\[4pt]
 \eg{Maximax $\to$ Stock \quad Maximin $\to$ Bond}}
