% ---------- VI2 Measuring risk (x=22.07, y=10.45) ----------
\pbox{22.07}{10.45}{6.47}{8.25}{ured}{%
 \hd{ured}{2. Measuring Risk}
 Risk is measured by SD and CV.
 \frm{Expected return $\bar{X}=\Sigma\,X_i\,P(X_i)$\\[3pt]
  Variance $\sigma^2=\Sigma d^2/n$ \ (or $\Sigma d^2P$)\\[3pt]
  SD $\sigma=\sqrt{\sigma^2}$, \ $d=X_i-\bar{X}$\\[3pt]
  CV $=\sigma/\bar{X}\times100$}{22pt}
 \eg{Eg: 12, 18, 20, 15, 25 (\%)\\ mean 18\%, $\sigma=4.43\%$ ($\div n$)\\ CV $=24.6\%$}}
\begin{scope}[shift={(22.07,10.45)},line width=3.5pt]
  \draw (0.3,0.7) -- (6.2,0.7);
  \draw[ugreen,line width=4.5pt,smooth] plot[domain=-1.7:1.7,samples=70] ({3.25+\x},{0.7+2.6*exp(-(\x*\x)/0.5)});
  \draw[ured,line width=4.5pt,smooth] plot[domain=-2.9:2.9,samples=80] ({3.25+\x},{0.7+1.182*exp(-(\x*\x)/2.42)});
  \draw[dashed,thin] (3.25,0.7) -- (3.25,3.5);
  \node[font=\lblsize] at (3.25,0.35) {$\bar{X}$ (mean)};
  \node[font=\fontsize{22}{26}\selectfont,ugreen,anchor=west] at (3.95,3.0) {low $\sigma$: stable};
  \node[font=\fontsize{22}{26}\selectfont,ured,anchor=west] at (0.35,1.9) {high $\sigma$:};
  \node[font=\fontsize{22}{26}\selectfont,ured,anchor=west] at (0.35,1.6) {risky};
\end{scope}
