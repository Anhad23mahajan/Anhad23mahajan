% V2 Uneven cash flows   (x=9.35)
\pbox{9.35}{10.45}{8.35}{8.25}{ublue}{%
 \hd{ublue}{2. Uneven Cash Flows}
 Unequal cash flows: discount each to today.
 \frm{NPV $=\Sigma\,\dfrac{C_t}{(1+k)^t}-I$}{50pt}
 {\smallsize $C_t$ = cash flow, $k$ = discount rate, $I$ = investment}\\[2pt]
 NPV $>0$: accept; NPV $<0$: reject.\\
 IRR = rate where NPV $=0$; accept if IRR $>k$.\\[2pt]
 \eg{Eg: $I=$ Rs 1,00,000, $k=10\%$\\ NPV $\approx$ Rs 27,340 $>0$, so accept}\\
 {\smallsize (PVIF 0.909, 0.826, 0.751)}}
